import cv2
import numpy as np
import matplotlib.pyplot as plt

# Helper function to display images in Jupyter/Colab notebooks
def show_image(img, title="Image"):
    plt.figure(figsize=(6, 6))
    # OpenCV uses BGR, Matplotlib uses RGB. We must convert before displaying.
    plt.imshow(cv2.cvtColor(img, cv2.COLOR_BGR2RGB))
    plt.title(title)
    plt.axis('off')
    plt.show()
# Create a black/dark gray canvas (500x500 pixels, 3 color channels)
img = np.zeros((500, 500, 3), dtype=np.uint8) + 40 

# 1. Draw a Red Triangle (Color BGR: Blue=0, Green=0, Red=255)
pts = np.array([[100, 100], [50, 200], [150, 200]], np.int32)
pts = pts.reshape((-1, 1, 2))
cv2.fillPoly(img, [pts], (0, 0, 255))

# 2. Draw a Green Square (Color BGR: Blue=0, Green=255, Red=0)
cv2.rectangle(img, (300, 100), (420, 220), (0, 255, 0), -1)

# 3. Draw a Blue Circle (Color BGR: Blue=255, Green=0, Red=0)
cv2.circle(img, (250, 380), 60, (255, 0, 0), -1)

# Show the generated image
show_image(img, "Original Synthetic Image")
# Copy the image so we don't draw over our original
output_img = img.copy()

# Convert to grayscale and threshold to binary (black and white) image
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
_, thresh = cv2.threshold(gray, 50, 255, cv2.THRESH_BINARY)

# Find contours (boundaries) in the thresholded image
contours, _ = cv2.findContours(thresh, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

for contour in contours:
    # Calculate the area to filter out tiny noise particles
    area = cv2.contourArea(contour)
    if area < 100:
        continue
        
    # 1. Identify Geometric Shape
    # Approximate the polygon curve with epsilon accuracy
    perimeter = cv2.arcLength(contour, True)
    approx = cv2.approxPolyDP(contour, 0.04 * perimeter, True)
    num_vertices = len(approx)
    
    shape_name = "Unknown"
    if num_vertices == 3:
        shape_name = "Triangle"
    elif num_vertices == 4:
        shape_name = "Square"
    else:
        shape_name = "Circle"
        
    # 2. Identify Color
    # Get the center (centroid) of the shape using image moments
    M = cv2.moments(contour)
    if M["m00"] != 0:
        cX = int(M["m10"] / M["m00"])
        cY = int(M["m01"] / M["m00"])
    else:
        cX, cY = 0, 0
        
    # Sample the color from the original image at the center pixel
    # Remember BGR format in OpenCV
    bgr_pixel = img[cY, cX]
    b, g, r = bgr_pixel[0], bgr_pixel[1], bgr_pixel[2]
    
    color_name = "Unknown"
    if r > g and r > b:
        color_name = "Red"
    elif g > r and g > b:
        color_name = "Green"
    elif b > r and b > g:
        color_name = "Blue"
        
    # Label to put on the image
    label = f"{color_name} {shape_name}"
    
    # Draw the contour outline and write the label on the image
    cv2.drawContours(output_img, [contour], -1, (255, 255, 255), 2)
    cv2.putText(output_img, label, (cX - 50, cY), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 2)
    
    print(f"Detected Object: {label} at center ({cX}, {cY})")

# Display final recognized output
show_image(output_img, "Recognized Features & Objects")ss
